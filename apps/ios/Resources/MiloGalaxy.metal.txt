#include <metal_stdlib>
using namespace metal;

// Faithful port of apps/web/src/components/blackHoleGalaxyShader.ts and
// BLACK_HOLE_GALAXY_SETTINGS. Keep six layer populations, seeds, depth and
// clockwise GLSL column-major rotation in sync with that reference.
constant float uStarSpeed = 0.7;
constant float uDensity = 1.7;
constant float uHueShift = 140.0;
constant float uSpeed = 1.4;
constant float uGlowIntensity = 0.45;
constant float uSaturation = 0.15;
constant float uRepulsionStrength = 2.0;
constant float uTwinkleIntensity = 0.4;
constant float uRotationSpeed = 0.1;
constant float STAR_SIZE_GAIN = 1.65;
constant float POINTER_REPULSION_SCALE = 0.055;
constant float MICRO_DUST_POPULATION = 0.16;
constant float FINE_STAR_POPULATION = 0.079;
constant float MID_STAR_POPULATION = 0.056;
constant float STARFIELD_DENSITY_FLOOR = 0.52;


struct MiloGalaxyField {
    float uTime;
    float uPointerActive;
float hash21(float2 p) {
  p = fract(p * float2(123.34, 456.21));
  p += dot(p, p + 45.32);
  return fract(p.x * p.y);
}

float noise2(float2 p) {
  float2 i = floor(p);
  float2 f = fract(p);
  f = f * f * (3.0 - 2.0 * f);
  return mix(
    mix(hash21(i), hash21(i + float2(1.0, 0.0)), f.x),
    mix(hash21(i + float2(0.0, 1.0)), hash21(i + float2(1.0)), f.x),
    f.y
  );
}

float fbm(float2 p) {
  float value = 0.0;
  float amplitude = 0.55;
  for (int octave = 0; octave < 5; octave++) {
    value += noise2(p) * amplitude;
    p = p * 2.03 + float2(17.2, 9.4);
    amplitude *= 0.5;
  }
  return value;
}

float2x2 rotate2d(float angle) {
  float s = sin(angle);
  float c = cos(angle);
  return float2x2(c, -s, s, c);
}

float3 hueColor(float hue) {
  return 0.5 + 0.5 * cos(6.2831853 * (hue + float3(0.0, 0.333, 0.667)));
}

float2 repelPointer(float2 uv, float2 pointer) {
  float2 delta = uv - pointer;
  float distanceToPointer = max(length(delta), 0.025);
  float influence = exp(-distanceToPointer * 8.0) * uPointerActive;
  return uv + delta / max(length(delta), 0.0001) * influence * uRepulsionStrength * POINTER_REPULSION_SCALE;
}

float starLayer(
  float2 uv,
  float scale,
  float layerSeed,
  float population,
  float2 radiusRange
) {
  float2 grid = uv * scale + layerSeed;
  float2 cell = floor(grid);
  float2 local = fract(grid) - 0.5;
  float seed = hash21(cell + layerSeed);
  float2 offset = float2(seed, hash21(cell + 7.31)) - 0.5;
  float driftTime = uTime * uSpeed;
  float2 stellarDrift = float2(
    sin(driftTime * (0.12 + seed * 0.05) + seed * 19.2),
    cos(driftTime * (0.10 + seed * 0.04) + seed * 23.7)
  ) * 0.018;
  float distanceToStar = length(local - offset * 0.58 - stellarDrift);
  float radius = mix(radiusRange.x, radiusRange.y, seed) * STAR_SIZE_GAIN;
  float starCore = 1.0 - smoothstep(radius * 0.24, radius * 0.78, distanceToStar);
  float starHalo = 1.0 - smoothstep(radius * 0.62, radius * 2.25, distanceToStar);
  float star = starCore + starHalo * 0.22;
  float threshold = 1.0 - clamp(population * uDensity, 0.0, 0.62);
  float twinkle = 1.0 + sin(uTime * (1.15 + seed * 2.6) + seed * 40.0) * uTwinkleIntensity;
  return star * step(threshold, seed) * twinkle;
}

float galaxyDepth(float phase) {
  return fract(phase + uTime * uStarSpeed * uSpeed * 0.022);
}

float galaxyDepthFade(float depth) {
  float fadeIn = smoothstep(0.0, 0.14, depth);
  float fadeOut = 1.0 - smoothstep(0.82, 1.0, depth);
  return fadeIn * fadeOut;
}

float animatedStarLayer(
  float2 uv,
  float scale,
  float layerSeed,
  float population,
  float2 radiusRange,
  float phase
) {
  float depth = galaxyDepth(phase);
  float zoom = mix(0.84, 1.24, depth);
  float depthGain = mix(0.72, 1.18, depth) * galaxyDepthFade(depth);
  float2 layerDrift = float2(
    sin(uTime * uSpeed * 0.055 + layerSeed),
    cos(uTime * uSpeed * 0.043 + layerSeed * 1.37)
  ) * 0.006;
  return starLayer(uv * zoom + layerDrift, scale, layerSeed, population, radiusRange) * depthGain;
}

float microStarDust(float2 uv) {
  float dust = animatedStarLayer(
    uv + float2(-0.19, 0.27),
    420.0,
    73.6,
    MICRO_DUST_POPULATION,
    float2(0.20, 0.38),
    0.0
  ) * 0.20;
  dust += animatedStarLayer(
    uv + float2(0.37, -0.16),
    332.0,
    91.8,
    MICRO_DUST_POPULATION * 0.82,
    float2(0.18, 0.34),
    0.5
  ) * 0.24;
  return dust;
}

float layeredStarField(float2 uv) {
  float clusterA = smoothstep(0.28, 0.74, fbm(uv * 1.72 + float2(8.4, -3.7)));
  float clusterB = smoothstep(
    0.40,
    0.76,
    fbm(rotate2d(-0.34) * uv * 3.15 + float2(-4.8, 11.2))
  );
  float densityFloor = STARFIELD_DENSITY_FLOOR;
  float cluster = densityFloor + clusterA * 0.54 + clusterB * 0.31;

  float stars = microStarDust(uv) * mix(0.78, 1.22, clusterB);
  stars += animatedStarLayer(
    uv,
    214.0,
    41.2,
    FINE_STAR_POPULATION,
    float2(0.13, 0.27),
    0.1667
  ) * 0.25;
  stars += animatedStarLayer(
    uv + float2(0.13, -0.21),
    132.0,
    19.4,
    MID_STAR_POPULATION,
    float2(0.10, 0.23),
    0.3333
  ) * 0.42;
  stars += animatedStarLayer(
    uv - float2(0.31, 0.07),
    72.0,
    7.7,
    0.038,
    float2(0.07, 0.18),
    0.6667
  ) * 0.72;
  stars += animatedStarLayer(
    uv + float2(0.08, 0.11),
    34.0,
    2.7,
    0.010,
    float2(0.045, 0.12),
    0.8333
  ) * 1.24;
  return stars * cluster;
}


};

struct MiloGalaxyUniforms {
    float2 size;
    float2 drawableSize;
    float2 pointer;
    float time;
    float active;
};

vertex float4 miloGalaxyVertex(uint vertexID [[vertex_id]]) {
    const float2 vertices[3] = {float2(-1.0, -1.0), float2(3.0, -1.0), float2(-1.0, 3.0)};
    return float4(vertices[vertexID], 0.0, 1.0);
}

fragment half4 miloGalaxyFragment(float4 rasterPosition [[position]],
                                  constant MiloGalaxyUniforms &uniforms [[buffer(0)]]) {
    float2 size = uniforms.size;
    float time = uniforms.time;
    float2 pointerPosition = uniforms.pointer;
    float pointerActive = uniforms.active;
    float2 position = rasterPosition.xy / max(uniforms.drawableSize, float2(1.0)) * size;
    MiloGalaxyField field = {time, pointerActive};
    // Web caps its drawing buffer at 1.5 pixels/point. Sample on that same grid.
    float2 point = (floor(position * 1.5) + 0.5) / 1.5;
    float2 normalized = point / max(size, float2(1.0));
    float2 vUv = float2(normalized.x, 1.0 - normalized.y);
    float2 scale = float2(size.x / max(size.y, 1.0), 1.0);
    float2 uv = (vUv * 2.0 - 1.0) * scale;
    float2 pointer = (pointerPosition * 2.0 - 1.0) * scale;
    float2 starUv = field.rotate2d(time * uRotationSpeed * 0.18) * uv;
    float stars = field.layeredStarField(field.repelPointer(starUv, pointer));
    float3 coldTint = field.hueColor(fract(uHueShift / 360.0));
    float3 silver = mix(float3(0.82, 0.85, 0.91), coldTint, uSaturation * 0.08);
    float3 color = silver * stars * (0.82 + uGlowIntensity * 0.88);
    float alpha = clamp(max(max(color.r, color.g), color.b) * 1.38, 0.0, 1.0);
    // Web clears transparent, blends SRC_ALPHA into RGB and alpha, then the
    // non-premultiplied canvas composites onto black: RGB * alpha^3.
    color = clamp(color * alpha, 0.0, 1.0) * alpha * alpha;
    // Same foreground vignette as .stage--black-hole-demo::after.
    float vertical = normalized.y < 0.34
        ? mix(0.12, 0.0, normalized.y / 0.34)
        : mix(0.0, 0.2, (normalized.y - 0.34) / 0.66);
    float cornerRadius = length(float2(size.x * 0.5, size.y * 0.53));
    float radialPosition = length(point - size * float2(0.5, 0.47)) / max(cornerRadius, 1.0);
    float radial = clamp((radialPosition - 0.26) / (0.72 - 0.26), 0.0, 1.0) * 0.16;
    return half4(half3(color * (1.0 - vertical) * (1.0 - radial)), 1.0);
}
